## 从零实现 GRU

```python
def get_params(vocab_size, num_hiddens, device):
    num_inputs = num_outputs = vocab_size

    def normal(shape):
        return torch.randn(size=shape, device=device) * 0.01

    def three():
        # 返回 (W_x, W_h, b)，分别对应输入项、隐状态项、偏置项
        # W_x: (vocab_size, num_hiddens), W_h: (num_hiddens, num_hiddens)
        return (normal((num_inputs, num_hiddens)),
                normal((num_hiddens, num_hiddens)),
                torch.zeros(num_hiddens, device=device))

    # 更新门 z_t
    W_xz, W_hz, b_z = three()
    # 重置门 r_t
    W_xr, W_hr, b_r = three()
    # 候选隐状态 h~_t
    W_xh, W_hh, b_h = three()

    # 输出层
    W_hq = normal((num_hiddens, num_outputs))
    b_q = torch.zeros(num_outputs, device=device)

    params = [W_xz, W_hz, b_z, W_xr, W_hr, b_r, W_xh, W_hh, b_h, W_hq, b_q]
    for p in params:
        p.requires_grad_(True)
    return params

def init_gru_state(batch_size, num_hiddens, device):
    # H: (batch_size, num_hiddens)
    return (torch.zeros((batch_size, num_hiddens), device=device),)

def gru(inputs, state, params):
    """
    inputs: (num_steps, batch_size, vocab_size)
    state: (H,), H shape=(batch_size, num_hiddens)
    """
    W_xz, W_hz, b_z, W_xr, W_hr, b_r, W_xh, W_hh, b_h, W_hq, b_q = params
    H, = state
    outputs = []

    for X in inputs:  # X: (batch_size, vocab_size)
        Z = torch.sigmoid(X @ W_xz + H @ W_hz + b_z)
        R = torch.sigmoid(X @ W_xr + H @ W_hr + b_r)
        H_tilda = torch.tanh(X @ W_xh + (R * H) @ W_hh + b_h)
        H = Z * H + (1 - Z) * H_tilda
        Y = H @ W_hq + b_q
        outputs.append(Y)

    return torch.cat(outputs, dim=0), (H,)
```

## 简洁实现（nn.GRU）

```python
class RNNModel(nn.Module):
    """适配 train_ch8 的高级 API RNN 包装器。"""
    def __init__(self, rnn_layer, vocab_size):
        super().__init__()
        self.rnn = rnn_layer
        self.vocab_size = vocab_size
        self.num_hiddens = rnn_layer.hidden_size
        self.num_directions = 2 if rnn_layer.bidirectional else 1
        self.linear = nn.Linear(self.num_hiddens * self.num_directions, vocab_size)

    def forward(self, inputs, state):
        # inputs: (batch_size, num_steps) -> one-hot -> (num_steps, batch_size, vocab_size)
        X = F.one_hot(inputs.T.long(), self.vocab_size).to(torch.float32)
        Y, state = self.rnn(X, state)
        # Y: (num_steps, batch_size, hidden) -> flatten 时间与批量维
        output = self.linear(Y.reshape((-1, Y.shape[-1])))
        return output, state

    def begin_state(self, batch_size, device):
        num_layers = self.rnn.num_layers * self.num_directions
        if not isinstance(self.rnn, nn.LSTM):
            return torch.zeros((num_layers, batch_size, self.num_hiddens), device=device)
        return (torch.zeros((num_layers, batch_size, self.num_hiddens), device=device),
                torch.zeros((num_layers, batch_size, self.num_hiddens), device=device))

gru_layer = nn.GRU(input_size=vocab_size, hidden_size=num_hiddens)
model = RNNModel(gru_layer, vocab_size).to(device)
train_ch8(model, train_iter, vocab, lr, num_epochs, device)
```
